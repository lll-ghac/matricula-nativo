**Opción B: Calcular DV automáticamente en Backend**

**Archivo:** `backend/routes/students.supabase.js` o nuevo `utils/rut.js`

```javascript
/**
 * Calcula el dígito verificador de un RUT chileno
 * Para RUTs extranjeros (>=100000000), retorna '0'
 */
function calculateDV(rut) {
  // RUT extranjero: usar '0' como DV
  if (rut >= 100000000) {
    return '0';
  }

  // Algoritmo de cálculo de DV chileno
  let suma = 0;
  let multiplo = 2;

  while (rut > 0) {
    suma += (rut % 10) * multiplo;
    rut = Math.floor(rut / 10);
    multiplo = multiplo === 7 ? 2 : multiplo + 1;
  }

  const resto = suma % 11;
  const dv = 11 - resto;

  if (dv === 11) return '0';
  if (dv === 10) return 'K';
  return dv.toString();
}

/**
 * Normaliza datos del alumno antes de insertar
 */
function normalizarAlumno(alumno) {
  // Si no tiene DV, calcularlo
  if (!alumno.dv && alumno.rut) {
    alumno.dv = calculateDV(alumno.rut);
  }

  return alumno;
}
```

    errores.push('RUT es obligatorio');

  return errores;

/**
 * Valida datos completos del apoderado antes de insertar
 */
function validarDatosApoderado(apoderado) {
  const camposRequeridos = [
    'tipo_documento', 'nombre', 'apellido_paterno',
    'apellido_materno', 'genero', 'fecha_nacimiento',
    'nacionalidad', 'direccion', 'celular'
  ];

  const faltantes = camposRequeridos.filter(campo => {
    const valor = apoderado[campo];
    return !valor || (typeof valor === 'string' && valor.trim() === '');
  });

  if (faltantes.length > 0) {
    throw new Error(`Campos obligatorios faltantes en apoderado: ${faltantes.join(', ')}`);
  }

  // Validar documento según tipo
  switch(apoderado.tipo_documento) {
    case 'RUT':
      if (!apoderado.rut || !apoderado.dv) {
        throw new Error('RUT y DV son obligatorios para tipo_documento RUT');
      }
      break;
    case 'DNI':
      if (!apoderado.dni) {
        throw new Error('DNI es obligatorio para tipo_documento DNI');
      }
      break;
    case 'Pasaporte':
      if (!apoderado.pasaporte) {
        throw new Error('Pasaporte es obligatorio para tipo_documento Pasaporte');
      }
      break;
  }

  return true;
}

/**
 * Validar alumno antes de insertar
 */
function validarDatosAlumno(alumno) {
  const camposRequeridos = [
    'rut', 'nombres', 'apellido_paterno', 'apellido_materno',
    'fecha_nacimiento', 'genero', 'nacionalidad', 'direccion',
    'region', 'comuna', 'con_quien_vive'
  ];

  const faltantes = camposRequeridos.filter(campo => {
    const valor = alumno[campo];
    return !valor || (typeof valor === 'string' && valor.trim() === '');
  });

  // DV obligatorio solo para chilenos
  if (alumno.nacionalidad === 'Chilena' && !alumno.dv) {
    faltantes.push('dv');
  }

  if (faltantes.length > 0) {
    throw new Error(`Campos obligatorios faltantes en alumno: ${faltantes.join(', ')}`);
  }

  return true;
}
